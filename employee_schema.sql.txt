-- Employee CSV Exporter: database setup for MySQL
-- Run this script in MySQL Workbench or a MySQL command-line client.

CREATE DATABASE IF NOT EXISTS employee_db;
USE employee_db;

CREATE TABLE IF NOT EXISTS employees (
    employee_id INT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    department VARCHAR(80) NOT NULL,
    email VARCHAR(120) NOT NULL UNIQUE,
    salary DECIMAL(10, 2) NOT NULL
);

-- Insert example records. INSERT IGNORE skips rows that conflict with an existing
-- employee ID or email, so re-running this script does not add duplicates.
INSERT IGNORE INTO employees (employee_id, full_name, department, email, salary)
VALUES

-- Verify the sample data (optional).
SELECT employee_id, full_name, department, email, salary
